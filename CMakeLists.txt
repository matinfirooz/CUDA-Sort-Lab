cmake_minimum_required(VERSION 3.24)
project(cuda_sort_lab VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()

add_executable(cpu_tests tests/cpu_tests.cpp)
target_include_directories(cpu_tests PRIVATE include)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(cpu_tests PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME cpu_reference_checks COMMAND cpu_tests)

find_package(Python3 QUIET COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
    add_test(NAME algorithm_models COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_models.py)
endif()

option(CUDA_SORT_ENABLE_CUDA "Build CUDA library and benchmark executable" ON)
option(CUDA_SORT_REGISTER_GPU_TESTS "Add GPU regression to CTest (requires a GPU at runtime)" OFF)
if(CUDA_SORT_ENABLE_CUDA)
    include(CheckLanguage)
    check_language(CUDA)
    if(CMAKE_CUDA_COMPILER)
        # An explicit value is portable to build servers with no GPU. Override
        # it for your target, e.g. -DCMAKE_CUDA_ARCHITECTURES=75 or 86 or 89.
        if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
            set(CMAKE_CUDA_ARCHITECTURES 75 CACHE STRING "CUDA target architectures")
        endif()
        enable_language(CUDA)
        find_package(CUDAToolkit 12.0 REQUIRED)
        set(CMAKE_CUDA_STANDARD 17)
        set(CMAKE_CUDA_STANDARD_REQUIRED ON)

        add_library(cudasort STATIC src/sorts.cu)
        target_include_directories(cudasort PUBLIC include)
        target_link_libraries(cudasort PUBLIC CUDA::cudart_static)
        add_executable(cuda_sorts src/main.cu)
        target_link_libraries(cuda_sorts PRIVATE cudasort)
        add_executable(sort_example examples/basic.cu)
        target_link_libraries(sort_example PRIVATE cudasort)
        # Runtime tests are opt-in so compile-only CI does not require a GPU.
        add_test(NAME cuda_cli_list COMMAND cuda_sorts --list)
        if(CUDA_SORT_REGISTER_GPU_TESTS)
            add_test(NAME gpu_correctness COMMAND cuda_sorts --self-test)
            set_tests_properties(gpu_correctness PROPERTIES TIMEOUT 600)
        endif()
    else()
        message(WARNING "CUDA compiler not found: building CPU checks only")
    endif()
endif()
